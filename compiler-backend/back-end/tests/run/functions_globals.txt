% Functions with several outputs, a global counter, loops and compound conditions
_divmod input parameter list [int b2, int b3]
output parameter list [int b4, int b5];
	b4 <--- b2 / b3;
	b5 <--- b2 - b4 * b3;
	return [b4, b5];
end

_gcd input parameter list [int b2, int b3]
output parameter list [int b4];
	type int : b5;
	type int : b6;
	type int : b7;
	b5 <--- b2;
	b6 <--- b3;
	while (b6 != 0)
		[b7, b4] <--- call _divmod with parameters [b5, b6];
		b5 <--- b6;
		b6 <--- b4;
		c2 <--- c2 + 1;
	endwhile
	b4 <--- b5;
	return [b4];
end

_main
	type int : b2;
	type int : b3;
	type int : b4;
	type int : b5;
	type int : c2 : global;
	read(b2);
	read(b3);
	[b4, b5] <--- call _divmod with parameters [b2, b3];
	write(b4);
	write(b5);
	[b4] <--- call _gcd with parameters [b2, b3];
	write(b4);
	write(c2);
	if ((b2 > 100) @@@ (~(b3 < 5))) then
		write(1);
	else
		write(0);
	endif
	return;
end
